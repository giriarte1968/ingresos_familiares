# Import: from parsers.mercado_inmobiliario import valuar_propiedad, actualizar_base_ciudad_web

def valuar_propiedad(propiedad, fecha_ref=None):
    """
    Valua una propiedad usando el MODELO DESACOPLADO AVANZADO (v3.0).

    Calcula:
    - Valor m2 usando base_ciudad * indice_mercado * factores (zona, estado, calidad, piso, orientacion, amenities, etc.)
    - Valor total = valor_m2 * m2
    - Serie historica mensual con interpolacion + suavizado
    - Plusvalia mensual y acumulada
    - Tendencia (alcista/bajista/neutral)

    Args:
        propiedad: dict con datos de la propiedad (zona, m2, tipo_inmueble, estado_detalle, etc.)
        fecha_ref: string "YYYY-MM" o datetime para la fecha de referencia (default: mes actual)

    Returns:
        dict = {
            'valor_m2_actual_usd': float,
            'rango_m2': str (ej: "USD 1,200 - 1,400"),
            'valor_propiedad_usd': float,
            'serie_mensual_m2': [{'fecha': str, 'valor': float, 'fuente': str}, ...],
            'plusvalia_mensual_pct': float,
            'plusvalia_acumulada_pct': float,
            'tendencia': str (alcista/bajista/neutral),
            'factores_aplicados': dict,
            'nivel_confianza': str (alto/medio/bajo),
            'justificacion': str
        }
    """
